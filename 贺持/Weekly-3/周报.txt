一、欧拉角与四元数
	欧拉角：用三个独立的角度（x、y、z）表示绕对应轴的旋转，单位是度（°）。 在 Unity 里，transform.eulerAngles 就是物体的欧拉角旋转值。
	四元数：用四个浮点数（x、y、z、w）表示旋转，本质是复数的扩展，数学上基于三维旋转的“轴角”表示法。在 Unity 里，transform.rotation 就是物体的四元数旋转值。

	二者转换：
	欧拉角--->四元数
	// 绕Y轴旋转90度的四元数 
	Quaternion rot = Quaternion.Euler(0, 90, 0); 
	// 给物体设置旋转 
	transform.rotation = rot;

	四元数--->欧拉角
	Quaternion rot = transform.rotation; 
	Vector3 euler = rot.eulerAngles; // 得到对应的欧拉角(x,y,z) 
	Debug.Log(euler.y); // 查看Y轴的旋转角度

四、同步与异步加载场景
	创建新的场景：Project--->右键--->Creat--->Scene
	场景转换：File--->Building Settings--->把两个场景拖进来

	同步加载场景代码：
using System.Collections;using System.Collections.Generic;using UnityEngine;using UnityEngine.SceneManagement;
public class SceneTest : MonoBehaviour
{
    void Start()
    {
        //两个类，场景类，场景管理类
        //SceneManager.LoadScene(1);//同步加载，表示直接加载到1号场景
        //也可以名称跳转，如：
        //SceneManager.LoadScene("MyScene");
        //获取当前场景
        Scene scene = SceneManager.GetActiveScene();
        //场景名称
        Debug.Log(scene.name);
        //场景是否已经加载
        Debug.Log(scene.isLoaded);
        //场景路径
        Debug.Log(scene.path);
        //该场景在Building Settings里面对应的索引
        Debug.Log(scene.buildIndex);
        //获取当前场景所有游戏物体
        GameObject[] gos=scene.GetRootGameObjects();
        Debug.Log(gos.Length);

        //场景管理类
        //获取当前已加载场景数量
        Debug.Log(SceneManager.sceneCount);
        //创建新场景
        Scene newScene=SceneManager.CreateScene("newScene");
        //获取当前已加载场景数量
        Debug.Log(SceneManager.sceneCount);
        //卸载场景（异步销毁）
        SceneManager.UnloadSceneAsync(newScene);
        //加载场景
        //LoadSceneMode.Single加载方式作为替换存在
        //SceneManager.LoadScene("MyScene",LoadSceneMode.Single);
        //LoadSceneMode.Additive加载方式作为叠加存在，但加载特别大的场景时会卡顿
        SceneManager.LoadScene("MyScene",LoadSceneMode.Additive);
    }
    void Update()
    {
    }
}

	异步加载场景代码
using System.Collections;using System.Collections.Generic;using UnityEngine;using UnityEngine.SceneManagement;public class AsyncTest : MonoBehaviour
{
    AsyncOperation operation;
    void Start()
    {
        StartCoroutine(loadScene());
    }

    //协程方法用来异步加载场景
    IEnumerator loadScene()
    {
        operation=SceneManager.LoadSceneAsync(1);
        //加载完场景不要自动跳转
        operation.allowSceneActivation = false;
        yield return operation;
    }

    float timer = 0;

    void Update()
    {
        //输出加载进度    0-0.9
        Debug.Log(operation.progress);
        timer += Time.deltaTime;
        //如果到达5秒，再跳转
        if (timer > 5)
        {
            operation.allowSceneActivation = true;
        }
    }
}

七、虚拟轴
	定义：虚拟轴是一个数值在-1~1内的数轴，这个数轴上重要的数值就是-1、0和1。当使用按键模拟一个完整的虚拟轴时需要用到两个按键，即将按键1设置为负轴按键，按键2设置为正轴按键。在无任何按键时，虚拟轴数值为0；按下按键1时，虚拟轴数值会从0~-1进行过渡；按下按键2时，虚拟轴数值会从0~1过渡。
