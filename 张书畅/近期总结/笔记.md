\## 一、Git

\### 基础操作流程

```bash

\# 进入项目文件夹

cd D:/code/demo

\# 初始化git本地仓库，生成隐藏文件夹 .git

git init

\# 查看状态：查看哪些文件被修改、哪些未跟踪

git status

\# 将指定文件加入暂存区

git add index.html

\# 将所有修改文件加入暂存区（常用）

git add .

\# 提交到本地仓库，-m后面写提交说明，必须写注释

git commit -m "完成HTML首页结构编写"

\# 查看提交日志，看所有提交记录、commit id

git log

\# 简洁版日志，一行展示一条记录

git log --oneline

\# 版本回退 --hard：工作区、暂存区、仓库全部回退到指定版本号

git reset --hard 提交ID

\# 回退到上一个版本

git reset --hard HEAD^

\# 回退2个版本

git reset --hard HEAD\~2

\# 查看本地所有分支，\*代表当前所在分支

git branch

\# 创建新分支

git branch feature

\# 切换分支

git checkout feature

\# 创建并立刻切换到新分支（简写）

git checkout -b feature

\# 切回主分支

git checkout main

\# 把feature分支合并到当前main分支

git merge feature

\# 将本地仓库关联GitHub远程仓库地址

git remote add origin https://github.com/用户名/仓库名.git

\# 查看已经关联的远程仓库

git remote -v

\# 把本地main分支推送到远程origin

git push -u origin main

\# 拉取远程最新代码到本地

git pull origin main

\# 克隆别人远程仓库到本地

git clone https://github.com/xxx/xxx.git
```

\## 四、MySQL学习笔记

```
\-- 查看所有数据库

show databases;

\-- 创建数据库

create database db\_test;

\-- 创建数据库，如果不存在才创建（防止报错）

create database if not exists db\_test;

\-- 删除数据库

drop database db\_test;

drop database if exists db\_test;

\-- 使用数据库（切换数据库）

use db\_test;

\-- 查看当前正在使用的数据库

select database();

\-- 查看当前库下所有表

show tables;

\-- 查看表结构

desc user;

\-- 创建表

create table user(

&#x20;   id int,

&#x20;   name varchar(20),

&#x20;   age int

);

\-- 删除表

drop table user;

drop table if exists user;

\-- 修改表名

alter table user rename to t\_user;

\-- 添加字段

alter table t\_user add column phone varchar(11);

\-- 修改字段类型

alter table t\_user modify column age tinyint;

\-- 修改字段名+类型

alter table t\_user change column phone tel char(11);

\-- 删除字段

alter table t\_user drop column tel;

\-- 新增数据 全字段

insert into user values(1,'张三',20);

\-- 指定字段新增

insert into user(name,age) values('李四',22);

\-- 批量新增

insert into user(name,age) values

('王五',18),

('赵六',25);

\-- 修改数据，一定要加where条件！否则全表更新

update user set age=21 where id=1;

\-- 多字段修改

update user set name='张三三',age=22 where id=1;

\-- 删除数据，加where条件，否则整张表数据清空

delete from user where id=2;

\-- 查询表全部数据

select \* from user;

\-- 查询指定字段

select id,name from user;

\-- 别名 as，as可以省略

select name as 姓名, age 年龄 from user;

\-- 去重 distinct

select distinct age from user;

\-- where 条件查询

\-- 条件运算符：>  <  >=  <=  =  !=

select \* from user where age > 20;

\-- and 并且

select \* from user where age>18 and age<30;

\-- or 或者

select \* from user where age<18 or age>40;

\-- null 判断：is null / is not null，不能用 =null

select \* from user where name is null;

\-- 模糊查询 like

\-- % 匹配任意多个字符；\_匹配单个字符

select \* from user where name like '张%';

select \* from user where name like '\_三';

\-- order by 排序  asc升序(默认)  desc降序

select \* from user order by age asc;

select \* from user order by age desc, id asc;

\-- limit 分页 limit 起始索引,条数；起始索引从0开始

select \* from user limit 0,2; -- 第1页，每页2条

\-- 创建表时设置外键

create table emp(

&#x20;   id int primary key auto\_increment,

&#x20;   name varchar(20),

&#x20;   dept\_id int,

&#x20;   foreign key(dept\_id) references dept(id)

);

\-- 隐式内连接

select \* from dept,emp where dept.id = emp.dept\_id;

\-- 显式内连接 inner join ... on 

select \* from dept inner join emp on dept.id = emp.dept\_id;
```

\## 六、Vue2

\### Vue2常用指令

1\. `{{ 表达式 }}` 插值表达式，渲染data数据，可以写简单运算

2\. `v‑bind:属性` 简写 `:属性`，绑定html属性（src、class、title）

3\. `v‑on:事件` 简写 `@事件`，绑定事件（click点击，input输入）

4\. `v‑if` 条件渲染，销毁/创建dom；`v‑show` 通过display控制显示隐藏

5\. `v‑for="(item,index) in 数组"` 循环渲染列表，必须加`:key`提升性能

6\. `v‑model` 表单双向绑定，input、select、textarea使用，视图和数据互相同步

\### Vue实例选项

\- `el`：挂载选择器，绑定页面哪个div由vue管理

\- `data()`：存放页面的数据，函数返回对象

\- `methods`：存放事件处理函数，函数内部使用`this.xxx`访问data里面的数据
