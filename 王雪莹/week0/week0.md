# Git常用命令

git status                     # 看状态
git add .                      # 加到暂存区
git commit -m " "      # 提交
git log --oneline --graph --all # 看历史

git switch -c feature/xxx      # 新建并切换分支
git merge feature/xxx          # 合并分支
git push                       # 推送
